\section{DeepSeek API 消费监控}

\paragraph{脚本位置} \texttt{hermes/scripts/deepseek-monitor.py}

\paragraph{用途} 查询 DeepSeek API 账户余额，支持阈值告警和 Token 用量统计。

\paragraph{前提条件}
\begin{itemize}
  \item \texttt{DEEPSEEK\_API\_KEY} 环境变量已设置
  \item \texttt{pip install requests}
\end{itemize}

\paragraph{用法}
\begin{verbatim}
  # 仅查询余额
  ./hermes/scripts/deepseek-monitor.py

  # 余额低于 10 CNY 时退出码为 2（可用于告警链）
  ./hermes/scripts/deepseek-monitor.py --alert 10

  # 从 Hermes session 日志累计 Token 用量
  ./hermes/scripts/deepseek-monitor.py --track ~/.hermes/sessions
\end{verbatim}

\paragraph{API 端点}
\begin{itemize}
  \item \texttt{GET https://api.deepseek.com/user/balance}
  \item 官方文档: \url{https://api-docs.deepseek.com/zh-cn/api/get-user-balance}
\end{itemize}

\paragraph{输出示例}
\begin{alltt}

    DeepSeek API 消费监控

    状态:       \twemoji{white_check_mark} 可用
    总余额:     55.07 CNY
    充值余额:   55.07 CNY
    赠金余额:   0.00 CNY
    查询时间:   2026-05-15 15:34:08

\end{alltt}

\paragraph{集成到 Hermes cron} 可用 \texttt{no\_agent=true} 模式定时运行：
\begin{itemize}
  \item 每天 9 点推送余额到飞书
  \item 余额低于 10 元时告警
\end{itemize}
