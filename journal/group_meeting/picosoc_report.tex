\begin{frame}{项目背景与技术目标}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{项目背景}
        \begin{itemize}
          \item 基于开源 PicoRV32 (RV32IMC) 构建完整 SoC
          \item 采用 AXI4-Lite 总线连接外设
          \item 目标：C 语言固件开发 + FPGA 原型验证
        \end{itemize}
      \end{block}

      \begin{exampleblock}{已完成目标 \checkmark}
        \begin{enumerate}
          \item SoC 系统搭建 (PicoRV32 + AXI4-Lite + 外设)
          \item C 语言固件工具链与编译流程
          \item AXI-Lite 外设驱动 (UART/SPI/BRAM/GPIO)
          \item FPGA 原型验证 (UART 输出, GPIO LED)
          \item 仿真验证 (SPI 驱动)
        \end{enumerate}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{关键组件}
        \begin{itemize}
          \item \textbf{CPU}：PicoRV32 (RV32IMC)
          \item \textbf{总线}：AXI4-Lite 1主4从
          \item \textbf{指令存储}：XIP SPI Flash
          \item \textbf{数据存储}：内部 SRAM
          \item \textbf{外设}：UART, SPI Master, GPIO (MMR)
          \item \textbf{加速器}：PCPI 乘除法器
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

\section{AXI4-Lite 总线与地址映射}
% ============================================

\begin{frame}{AXI4-Lite 总线架构}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{AXI4-Lite 特性}
        \begin{itemize}
          \item ARM AMBA 协议精简版
          \item 5 个独立通道 (AW, W, B, AR, R)
          \item 单次读写 (single-beat)，无突发
          \item 固定 32-bit 字对齐
          \item VALID/READY 双向流水握手
        \end{itemize}
      \end{block}

      \begin{exampleblock}{本系统拓扑}
        \begin{itemize}
          \item \textbf{1 主}：PicoRV32 (Native AXI Master)
          \item \textbf{4 从}：SPI Flash / SRAM / UART / MMRs
          \item 地址译码选择从设备

\section{C 固件与设备驱动}
% ============================================

\begin{frame}{C 语言固件开发环境}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{block}{工具链}
        \begin{itemize}
          \item \textbf{编译器}：riscv32-unknown-elf-gcc
          \item \textbf{链接脚本}：定制 linker script (.ld)
          \item \textbf{启动代码}：汇编启动 (crt0.S)
          \item \textbf{烧录}：SPI Flash 编程器
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{block}{内存布局}
        \begin{itemize}
          \item \texttt{.text}：XIP Flash (0x0100\_0000)
          \item \texttt{.rodata}：XIP Flash
          \item \texttt{.data}：SRAM (初始化数据)
          \item \texttt{.bss}：SRAM (零初始化)
          \item \texttt{.stack}：SRAM 顶部向下增长
          \item \texttt{.heap}：SRAM 底部向上增长 (可选)
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{8pt}
  \begin{exampleblock}{固件编译与加载流程}
    C 源码 $\xrightarrow{\text{gcc}}$ ELF $\xrightarrow{\text{objcopy}}$ BIN
    $\xrightarrow{\text{烧录器}}$ SPI Flash $\xrightarrow{\text{XIP}}$ PicoRV32 取指执行
  \end{exampleblock}
\end{frame}

\begin{frame}{设备驱动: UART}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{驱动 API}
        \begin{itemize}
          \item \texttt{uart\_init(baudrate)} — 初始化波特率
          \item \texttt{uart\_putc(char c)} — 发送单字符
          \item \texttt{uart\_puts(const char *s)} — 发送字符串
          \item \texttt{uart\_getc()} — 接收单字符 (阻塞)
          \item \texttt{uart\_printf(fmt, ...)} — 格式化输出
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{底层实现}
        \begin{itemize}
          \item 寄存器映射到固定地址\\
                {\footnotesize \texttt{\#define UART\_BASE 0x02000000}}
          \item 发送：写 DAT 寄存器 $\to$ TX 引脚
          \item 接收：轮询 RX 状态 $\to$ 读 DAT 寄存器
          \item 波特率：写 DIV 分频器
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{5pt}
  \begin{exampleblock}{验证方式}
    \begin{itemize}
      \item FPGA 上板：通过 FTDI USB-UART 桥接芯片输出到 PC 串口终端
      \item 测试输出：\texttt{"PicoSoC Boot OK. Hello World!"}
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{设备驱动: SPI Master}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{驱动 API}
        \begin{itemize}
          \item \texttt{spi\_init(mode, clk\_div)} — 配置 SPI 模式与速率
          \item \texttt{spi\_transfer(tx\_buf, rx\_buf, len)} — 全双工传输
          \item \texttt{spi\_cs\_select(n)} — 片选控制
          \item \texttt{spi\_write\_byte(byte)} — 单字节发送
          \item \texttt{spi\_read\_byte()} — 单字节接收
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{硬件特性}
        \begin{itemize}
          \item 支持 CPOL/CPHA 四种 SPI 模式
          \item 可配置时钟分频
          \item TX/RX 独立内存阵列
          \item 控制 FSM: IDLE $\to$ SHIFT $\to$ DONE
          \item 自动 CS 控制逻辑
          \item 支持多字节连续传输
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{5pt}
  \begin{exampleblock}{验证方式}
    \begin{itemize}
      \item 仿真验证：Testbench 驱动 SPI Slave 模型，比对 MISO/MOSI 数据
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{设备驱动: BRAM 与 GPIO}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{BRAM (SRAM) 访问}
        \begin{itemize}
          \item 直接通过 AXI-Lite 读写\\
                {\footnotesize 地址区间: 0x0000\_0000 — 0x00FF\_FFFF}
          \item C 指针直接操作：\\
                {\footnotesize \texttt{uint32\_t *p = (uint32\_t*)0x00000000;}}\\
                {\footnotesize \texttt{*p = 0xDEADBEEF;}}
          \item 用途：堆栈、全局变量、数据缓冲
          \item 性能：单周期 SRAM 访问 (无等待)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{GPIO 驱动}
        \begin{itemize}
          \item MMR 映射 GPIO 寄存器：\\
                {\footnotesize OUT (0x0200\_0200): 输出值}\\
                {\footnotesize DIR (0x0200\_0204): 方向 (1=输出)}\\
                {\footnotesize IN  (0x0200\_0208): 输入值}
          \item LED 控制：\\
                {\footnotesize \texttt{gpio\_set\_dir(0xFF);}}\\
                {\footnotesize \texttt{gpio\_write(0xAA); // 交替亮灭}}
          \item 支持位操作与批量读写
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
