%% === Title ===
\node[title] at (0,5.8) {TIADC 时钟偏斜校准 (Timing Skew Calibration)};

%% === Left: Clock groups and channel array ===
\node[draw, fill=cyan!6, dashed, inner sep=0.3cm] (chanbox) at (-6.5,2.2) {
\begin{tikzpicture}[scale=0.65, every node/.style={font=\tiny}]
  \node[draw, fill=orange!20, minimum width=0.4cm, minimum height=0.4cm] (c1) at (0,0) {$\phi_1$};
  \node[draw, fill=orange!30, minimum width=0.4cm, minimum height=0.4cm] (c2) at (0.8,0) {$\phi_2$};
  \node[draw, fill=orange!40, minimum width=0.4cm, minimum height=0.4cm] (c3) at (1.6,0) {$\phi_3$};
  \node[draw, fill=orange!50, minimum width=0.4cm, minimum height=0.4cm] (c4) at (2.4,0) {$\phi_4$};
  \draw[->, dashed] (0.4,0.3) -- (0.8,0.3) node[midway, above] {$T_s$};
  \node at (1.2,-0.5) {4组时钟相位依次错开};

  \node at (1.2,-1.7) {共64通道};
\end{tikzpicture}
};

%% === Middle: Cross-correlation computation ===
\node[block, fill=purple!10, right=2.5cm of chanbox] (corr) {\textbf{互相关计算}\\$R_k = A_o^{(k)} \cdot A_o^{(k+1)}$\\逐对相邻通道};
\node[block, fill=purple!15, below=0.5cm of corr] (accum) {\textbf{分组累加}\\$A_1\!\sim\!A_{16}$ (每组4对)\\每组对应一时钟边界};

\draw[path, thick] (chanbox.east) -- node[above]{$A_o[1..64]$} (corr.west);

%% === Top right: Average and Error ===
\node[block, fill=red!8, right=2.5cm of corr] (avgall) {\textbf{全局平均}\\$\displaystyle \bar{A} = \frac{1}{16}\sum_{m=1}^{16} A_m$};

\node[block, fill=red!12, below=0.5cm of avgall] (error) {\textbf{误差计算}\\$e^{(t)} = -\displaystyle\sum_{m} \frac{A_o^{(t-1+4m)} \cdot A_o^{(t+4m)}}{s} \times 4 + \bar{A}$\\$t=2,3,4$};

\draw[path] (corr.east) -- (avgall.west);
\draw[path] (accum.east) -| (error.north);
\draw[path] (avgall.south) -- node[right]{\scriptsize $\bar{A}$} (error.north);

%% === Bottom: LMS and VDL ===
\node[block, fill=blue!10, below=0.5cm of error] (lms) {\textbf{LMS 迭代更新}\\$\Delta t_k^{(n+1)} = \Delta t_k^{(n)} + u \cdot e^{(t)}$\\$u = 2^{9}$};
\node[block, fill=yellow!15, right=2.2cm of lms] (ctrl) {\textbf{控制字截位}\\$\widehat{\Delta t} = \lfloor \Delta t / 2^{24} \rfloor$\\$\times$ 步长 $10^{-14}$s};

\draw[path] (error.south) -- (lms.north);
\draw[path] (lms) -- (ctrl);

%% === Feedback: VDL to clock groups ===
\draw[fback] (ctrl.north) -- ++(0,0.5) -| node[pos=0.7, above]{\scriptsize VDL 延迟控制} (chanbox.north);
